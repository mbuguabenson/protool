import { MoneyMakerTab } from "@/components/tabs/money-maker-tab"
import { TrendingUp } from "lucide-react"

export function Dashboard() {
  const [tickHistory, setTickHistory] = useState<number[]>([])
  const [theme, setTheme] = useState<"light" | "dark">("dark")

  // Update as new ticks arrive from your API
  useEffect(() => {
    if (newTick) {
      setTickHistory(prev => [...prev, newTick.digit])
    }
  }, [newTick])

  return (
    <Tabs defaultValue="money-maker">
      <TabsList>
        <TabsTrigger value="money-maker">
          <TrendingUp className="w-4 h-4 mr-2" />
          Money Maker
        </TabsTrigger>
        {/* Other tabs */}
      </TabsList>

      <TabsContent value="money-maker">
        <MoneyMakerTab 
          theme={theme}
          recentDigits={tickHistory}
        />
      </TabsContent>
    </Tabs>
  )
}

STRATEGIES INCLUDED (6):
═══════════════════════════════════════════════════════════════════════════════

1. OVER/UNDER
   Range: 0-4 (Under) vs 5-9 (Over)
   Threshold: 55% WAIT, 60% READY, 70%+ TRADE NOW

2. EVEN/ODD
   Range: Even (0,2,4,6,8) vs Odd (1,3,5,7,9)
   Threshold: 7%+ deviation

3. RISE/FALL
   Analysis: Is next digit higher or lower?
   Threshold: 8%+ directional bias

4. MATCHES
   Target: Hottest digit (most frequent)
   Threshold: 15%+ frequency

5. DIFFERS
   Target: Coldest digit (least frequent)
   Threshold: ≤5% frequency

6. HIGH/LOW
   Range: 7-9 (High) vs 0-2 (Low)
   Threshold: 55%+ dominance
